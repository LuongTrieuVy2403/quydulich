const { google } = require('googleapis'); // ID Google Sheet của bạn const SPREADSHEET_ID = "11PVVu-vBtoQ-Rhip3wviDxtq7Vp-xdNytw1VwwmzWog"; const ADMIN_PASSWORD = "24032004"; function getSheetsClient() { const auth = new google.auth.GoogleAuth({ credentials: { client_email: process.env.GOOGLE_CLIENT_EMAIL, private_key: process.env.GOOGLE_PRIVATE_KEY ? process.env.GOOGLE_PRIVATE_KEY.replace(/\\n/g, '\n') : undefined, }, scopes: ['https://www.googleapis.com/auth/spreadsheets'], }); return google.sheets({ version: 'v4', auth }); } export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET,POST,OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') { return res.status(200).end(); } const { action } = req.query; const sheets = getSheetsClient(); try { // 1. Lấy Tổng Quan if (action === 'getSummary' && req.method === 'GET') { const response = await sheets.spreadsheets.values.get({ spreadsheetId: SPREADSHEET_ID, range: 'TongQuan!B1:B3', }); const rows = response.data.values || []; return res.status(200).json({ tongQuy: Number(rows[0]?.[0]) || 0, daChi: Number(rows[1]?.[0]) || 0, conLai: Number(rows[2]?.[0]) || 0, }); } // 2. Lấy danh sách thành viên (Cột A sheet ThanhVien) if (action === 'getMemberList' && req.method === 'GET') { const response = await sheets.spreadsheets.values.get({ spreadsheetId: SPREADSHEET_ID, range: 'ThanhVien!A2:A100', }); const rows = response.data.values || []; const members = rows.map(r => r[0]).filter(Boolean); return res.status(200).json(members); } // 3. Tra cứu số dư cá nhân if (action === 'getMemberBalance' && req.method === 'GET') { const { name } = req.query; const response = await sheets.spreadsheets.values.get({ spreadsheetId: SPREADSHEET_ID, range: 'ThanhVien!A2:D100', }); const rows = response.data.values || []; const keyword = (name || '').toLowerCase().trim(); for (let row of rows) { const fullName = row[0] ? row[0].toString().toLowerCase() : ''; if (fullName.includes(keyword)) { return res.status(200).json({ found: true, name: row[0], daDong: Number(row[1]) || 0, daDung: Number(row[2]) || 0, conDu: Number(row[3]) || 0, }); } } return res.status(200).json({ found: false }); } // 4. Lấy lịch sử chi tiêu if (action === 'getRecentExpenses' && req.method === 'GET') { const response = await sheets.spreadsheets.values.get({ spreadsheetId: SPREADSHEET_ID, range: 'LichSuChiTieu!A2:E200', }); const rows = response.data.values || []; const recent = rows.reverse().slice(0, 10).map(row => { let dateStr = row[0]; try { const d = new Date(row[0]); dateStr = `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')} ${String(d.getDate()).padStart(2, '0')}/${String(d.getMonth() + 1).padStart(2, '0')}`; } catch (e) {} return { time: dateStr, description: row[1], target: row[2], amount: Number(row[3]) || 0 }; }); return res.status(200).json(recent); } // 5. Thêm chi tiêu (Trừ tiền) if (action === 'addExpense' && req.method === 'POST') { const { password, description, targetList, amount, note } = req.body; if (password !== ADMIN_PASSWORD) { return res.status(400).json({ success: false, message: "Mật khẩu Admin không chính xác!" }); } const targetText = Array.isArray(targetList) ? targetList.join(', ') : targetList; await sheets.spreadsheets.values.append({ spreadsheetId: SPREADSHEET_ID, range: 'LichSuChiTieu!A:E', valueInputOption: 'USER_ENTERED', requestBody: { values: [[new Date().toISOString(), description, targetText, Number(amount), note || '']] } }); return res.status(200).json({ success: true, message: "Đã trừ tiền thành công!" }); } // 6. Cộng tiền đóng của thành viên if (action === 'addMemberContribution' && req.method === 'POST') { const { password, memberName, amount } = req.body; if (password !== ADMIN_PASSWORD) { return res.status(400).json({ success: false, message: "Mật khẩu Admin không chính xác!" }); } const response = await sheets.spreadsheets.values.get({ spreadsheetId: SPREADSHEET_ID, range: 'ThanhVien!A2:B100', }); const rows = response.data.values || []; let foundRowIndex = -1; for (let i = 0; i < rows.length; i++) { if (rows[i][0] && rows[i][0].toString().trim() === memberName.trim()) { foundRowIndex = i + 2; break; } } if (foundRowIndex === -1) { return res.status(400).json({ success: false, message: "Không tìm thấy thành viên trong danh sách!" }); } const currentAmount = Number(rows[foundRowIndex - 2][1]) || 0; const newTotal = currentAmount + Number(amount); await sheets.spreadsheets.values.update({ spreadsheetId: SPREADSHEET_ID, range: `ThanhVien!B${foundRowIndex}`, valueInputOption: 'USER_ENTERED', requestBody: { values: [[newTotal]] } }); return res.status(200).json({ success: true, message: "Đã nạp " + Number(amount).toLocaleString('vi-VN') + " đ cho " + memberName + " thành công!" }); } return res.status(404).json({ error: "API Endpoint không tồn tại" }); } catch (error) { console.error(error); return res.status(500).json({ success: false, message: "Lỗi Server: " + error.message }); } }
